วินัยจาก skill (ส่วนที่ 1) + sync skill-ai-naja (ส่วนที่ 4) Implementation Plan

เป้าหมาย: ใส่วินัยย่อจาก skill tb-* ลง system prompt ของทุก role เป็นค่าเริ่มต้น และเพิ่ม skill เต็มจาก skill-ai-naja เป็น option ที่เปิดต่อ role ได้ผ่าน config

สถาปัตยกรรม: ส่วนที่ 1 แก้เฉพาะข้อความใน src/prompts.ts (ไม่เปลี่ยน schema) ส่วนที่ 4 เพิ่ม script sync (scripts/sync-skills-lib.ts ที่เทสต์ได้ + scripts/sync-skills.ts ตัวรัน) ที่ copy skill ลง plugin team เดิม (skills/skills/), ต่อกฎ "Skills" ท้าย system prompt ใน buildQueryOptions เฉพาะ role ที่เปิด skill และตรวจชื่อ skill ตอน loadConfig

Tech Stack: TypeScript (ESM, tsx), zod, vitest, Node fs / child_process — ไม่มี dependency ใหม่

Design: docs/superpowers/specs/2026-10-29-team-skills-jira-quota-design.md (ส่วนที่ 1 และ 4) · ส่วนที่ 2 และ 3 อยู่ใน plan แยก

For agentic workers: REQUIRED SUB-SKILL: Use tb-subagent-driven-development (recommended) or tb-executing-plans to implement this plan task-by-task. Steps keep the - [ ] marker next to each checkbox for tracking.

Global Constraints

  • TypeScript ตาม style ของ repo: คอมเมนต์ภาษาไทย, ไม่มี dependency ใหม่, Node >=20.6
  • ข้อความ prompt ของ agent เขียนเป็นภาษาอังกฤษ (เหมือน prompt เดิม) ข้อความที่แสดงผู้ใช้เป็นภาษาไทย
  • เทสต์ด้วย vitest (tests/**/*.test.ts) ห้ามเรียก SDK หรือ GitHub จริงในเทสต์
  • ไม่เปลี่ยน JSON schema ใน src/schemas.ts
  • system prompt ของแต่ละ role ยาวขึ้นไม่เกิน 1,500 ตัวอักษรจากเดิม (เดิม: pm 3231, planning 1093, frontend 1073, backend 1085, qa 1891, security 1637)
  • prompt ใน SYSTEM_PROMPTS ต้องไม่มีคำว่า HTML และ Skill tool
  • ค่าเริ่มต้นทุก role skills: [] เหมือนเดิม — role ที่ไม่เปิด skill ต้องได้ system prompt เดิมทุกตัวอักษร
  • deny list ของการ sync: tb-subagent-driven-development, tb-executing-plans, tb-using-git-worktrees, tb-finishing-a-development-branch, tb-requesting-code-review, tb-read-jira, web-pow, handoff
  • ต้นทาง sync ตามลำดับ: --from <path> → env SKILL_AI_NAJA_DIR → ~/ai-skills → clone https://github.com/papawin1997/skill-ai-naja.git แบบ --depth 1 ลง temp (ห้ามอ่านจาก ~/.claude/skills/)
  • ไม่ลบหรือเปลี่ยนพฤติกรรมเดิมที่ไม่เกี่ยว ; ห้าม commit — ผู้ใช้ตรวจ diff และ commit เอง
  • ห้ามใช้ repo นี้เป็น --project ของ agent-team เอง

import * as fs from 'node:fs';
import * as os from 'node:os';
import * as path from 'node:path';
import { afterEach, beforeEach, describe, expect, it } from 'vitest';
import { formatResult, listSourceSkills, resolveSource, syncSkills } from '../scripts/sync-skills-lib';

let root: string;
let source: string;
let target: string;
let manifestFile: string;
let licensesDir: string;

const write = (file: string, content = 'x') => {
  fs.mkdirSync(path.dirname(file), { recursive: true });
  fs.writeFileSync(file, content);
};
const skill = (dir: string, name: string, body = `# ${name}`) => write(path.join(dir, name, 'SKILL.md'), body);
const run = (deny: string[] = []) => syncSkills({ sourceDir: source, targetDir: target, manifestFile, licensesDir, deny });

beforeEach(() => {
  root = fs.mkdtempSync(path.join(os.tmpdir(), 'sync-skills-'));
  source = path.join(root, 'src');
  target = path.join(root, 'plugin', 'skills');
  manifestFile = path.join(root, 'plugin', 'synced.json');
  licensesDir = path.join(root, 'plugin', 'licenses');
  fs.mkdirSync(source, { recursive: true });
});

afterEach(() => fs.rmSync(root, { recursive: true, force: true }));

describe('listSourceSkills', () => {
  it('เอาเฉพาะโฟลเดอร์ที่มี SKILL.md ไม่อยู่ใน deny และไม่ขึ้นต้นด้วยจุด เรียงตามชื่อ', () => {
    skill(source, 'b-skill');
    skill(source, 'a-skill');
    skill(source, 'denied');
    skill(source, '.hidden');
    write(path.join(source, 'licenses', 'MIT.txt'));
    write(path.join(source, 'README.md'));
    expect(listSourceSkills(source, ['denied'])).toEqual(['a-skill', 'b-skill']);
  });
});

describe('syncSkills', () => {
  it('copy ทั้งโฟลเดอร์ (รวมไฟล์ประกอบ) และจดชื่อลง manifest', () => {
    skill(source, 'tb-scrutinize');
    write(path.join(source, 'tb-scrutinize', 'scripts', 'helper.js'), 'h');
    const r = run();
    expect(r.added).toEqual(['tb-scrutinize']);
    expect(fs.readFileSync(path.join(target, 'tb-scrutinize', 'scripts', 'helper.js'), 'utf8')).toBe('h');
    expect(JSON.parse(fs.readFileSync(manifestFile, 'utf8'))).toEqual(['tb-scrutinize']);
  });

  it('ข้าม skill ใน deny', () => {
    skill(source, 'web-pow');
    skill(source, 'tb-scrutinize');
    run(['web-pow']);
    expect(fs.existsSync(path.join(target, 'web-pow'))).toBe(false);
    expect(fs.existsSync(path.join(target, 'tb-scrutinize', 'SKILL.md'))).toBe(true);
  });

  it('sync ซ้ำ: ทับทั้งโฟลเดอร์ ไฟล์ที่หายจากต้นทางก็หายจากปลายทาง และนับเป็น updated', () => {
    skill(source, 'tb-scrutinize');
    write(path.join(source, 'tb-scrutinize', 'old.md'));
    run();
    fs.rmSync(path.join(source, 'tb-scrutinize', 'old.md'));
    skill(source, 'tb-scrutinize', 'v2');
    const r = run();
    expect(r.updated).toEqual(['tb-scrutinize']);
    expect(fs.existsSync(path.join(target, 'tb-scrutinize', 'old.md'))).toBe(false);
    expect(fs.readFileSync(path.join(target, 'tb-scrutinize', 'SKILL.md'), 'utf8')).toBe('v2');
  });

  it('ลบเฉพาะ skill ที่เคย sync มาแล้วหายจากต้นทางหรือถูก deny', () => {
    skill(source, 'gone');
    skill(source, 'to-deny');
    skill(source, 'stay');
    run();
    fs.rmSync(path.join(source, 'gone'), { recursive: true });
    const r = run(['to-deny']);
    expect(r.removed.sort()).toEqual(['gone', 'to-deny']);
    expect(fs.existsSync(path.join(target, 'gone'))).toBe(false);
    expect(fs.existsSync(path.join(target, 'to-deny'))).toBe(false);
    expect(JSON.parse(fs.readFileSync(manifestFile, 'utf8'))).toEqual(['stay']);
  });

  it('ไม่แตะ skill ที่เขียนเองใน plugin และไม่ทับเมื่อชื่อซ้ำ (รายงาน skipped)', () => {
    skill(target, 'my-own', 'mine');
    skill(target, 'clash', 'mine');
    skill(source, 'clash', 'theirs');
    const r = run();
    expect(r.skipped).toEqual(['clash']);
    expect(fs.readFileSync(path.join(target, 'clash', 'SKILL.md'), 'utf8')).toBe('mine');
    expect(fs.readFileSync(path.join(target, 'my-own', 'SKILL.md'), 'utf8')).toBe('mine');
    expect(JSON.parse(fs.readFileSync(manifestFile, 'utf8'))).toEqual([]);
  });

  it('copy licenses/ ของต้นทางไปที่ licensesDir', () => {
    skill(source, 'a');
    write(path.join(source, 'licenses', 'superpowers-MIT.txt'), 'MIT');
    run();
    expect(fs.readFileSync(path.join(licensesDir, 'superpowers-MIT.txt'), 'utf8')).toBe('MIT');
  });
});

describe('resolveSource', () => {
  it('--from มาก่อน env และ ~/ai-skills', () => {
    const from = path.join(root, 'from');
    fs.mkdirSync(from);
    expect(resolveSource({ from, env: { SKILL_AI_NAJA_DIR: source }, home: root })).toBe(from);
  });

  it('ใช้ SKILL_AI_NAJA_DIR เมื่อไม่มี --from', () => {
    expect(resolveSource({ env: { SKILL_AI_NAJA_DIR: source }, home: root })).toBe(source);
  });

  it('ใช้ ~/ai-skills เมื่อไม่ระบุอะไร', () => {
    const local = path.join(root, 'ai-skills');
    fs.mkdirSync(local);
    expect(resolveSource({ env: {}, home: root })).toBe(local);
  });

  it('คืน null เมื่อไม่มีต้นทางในเครื่อง (ต้อง clone)', () => {
    expect(resolveSource({ env: {}, home: root })).toBeNull();
  });

  it('path ที่ระบุแต่ไม่มีอยู่ -> error ที่บอก path', () => {
    const missing = path.join(root, 'nope');
    expect(() => resolveSource({ from: missing, env: {}, home: root })).toThrow(missing);
  });
});

describe('formatResult', () => {
  it('สรุปทุกหมวดและเตือนให้ commit เอง', () => {
    const text = formatResult({ added: ['a'], updated: [], removed: ['b'], skipped: [] });
    expect(text).toContain('เพิ่ม (1): a');
    expect(text).toContain('อัปเดต (0): -');
    expect(text).toContain('ลบ (1): b');
    expect(text).toContain('commit เอง');

import * as fs from 'node:fs';
import * as path from 'node:path';

/** skills/sync.json */
export interface SyncConfig {
  repo: string;
  deny: string[];
}

export interface SyncOptions {
  sourceDir: string;
  /** skills/skills ของ plugin team */
  targetDir: string;
  /** skills/synced.json: ชื่อ skill ที่ sync มาครั้งก่อน (ใช้ตัดสินว่าลบ/ทับได้ไหม) */
  manifestFile: string;
  /** ปลายทางของ licenses/ จากต้นทาง */
  licensesDir: string;
  deny: readonly string[];
}

export interface SyncResult {
  added: string[];
  updated: string[];
  removed: string[];
  /** ชื่อซ้ำกับ skill ที่เขียนเองใน plugin จึงไม่ทับ */
  skipped: string[];
}

const hasSkill = (dir: string): boolean => fs.existsSync(path.join(dir, 'SKILL.md'));

/** ชื่อ skill ในต้นทาง: โฟลเดอร์ที่มี SKILL.md ไม่ขึ้นต้นด้วยจุด และไม่อยู่ใน deny */
export function listSourceSkills(sourceDir: string, deny: readonly string[]): string[] {
  return fs
    .readdirSync(sourceDir, { withFileTypes: true })
    .filter((e) => e.isDirectory() && !e.name.startsWith('.') && !deny.includes(e.name))
    .map((e) => e.name)
    .filter((name) => hasSkill(path.join(sourceDir, name)))
    .sort();
}

function readManifest(file: string): string[] {
  if (!fs.existsSync(file)) return [];
  const value: unknown = JSON.parse(fs.readFileSync(file, 'utf8'));
  return Array.isArray(value) ? value.filter((v): v is string => typeof v === 'string') : [];
}

export function syncSkills(opts: SyncOptions): SyncResult {
  const previous = readManifest(opts.manifestFile);
  const result: SyncResult = { added: [], updated: [], removed: [], skipped: [] };
  const synced: string[] = [];
  fs.mkdirSync(opts.targetDir, { recursive: true });

  for (const name of listSourceSkills(opts.sourceDir, opts.deny)) {
    const target = path.join(opts.targetDir, name);
    const exists = fs.existsSync(target);
    // มีอยู่แล้วแต่ไม่ได้มาจากการ sync = skill ที่ผู้ใช้เขียนเอง ห้ามทับ
    if (exists && !previous.includes(name)) {
      result.skipped.push(name);
      continue;
    }
    fs.rmSync(target, { recursive: true, force: true });
    fs.cpSync(path.join(opts.sourceDir, name), target, { recursive: true, dereference: true });
    (exists ? result.updated : result.added).push(name);
    synced.push(name);
  }

  // ลบเฉพาะตัวที่เคย sync มาแต่รอบนี้ไม่มีแล้ว (หายจากต้นทางหรือถูกเพิ่มเข้า deny)
  for (const name of previous) {
    if (synced.includes(name)) continue;
    fs.rmSync(path.join(opts.targetDir, name), { recursive: true, force: true });
    result.removed.push(name);
  }

  const licenses = path.join(opts.sourceDir, 'licenses');
  if (fs.existsSync(licenses)) {
    fs.rmSync(opts.licensesDir, { recursive: true, force: true });
    fs.cpSync(licenses, opts.licensesDir, { recursive: true, dereference: true });
  }

  fs.writeFileSync(opts.manifestFile, `${JSON.stringify(synced, null, 2)}\n`);
  return result;
}

/** โฟลเดอร์ต้นทางในเครื่อง: --from → SKILL_AI_NAJA_DIR → ~/ai-skills ; null = ต้อง clone */
export function resolveSource(input: { from?: string; env: NodeJS.ProcessEnv; home: string }): string | null {
  const explicit = input.from ?? input.env.SKILL_AI_NAJA_DIR;
  if (explicit) {
    if (!fs.existsSync(explicit)) throw new Error(`ไม่พบโฟลเดอร์ต้นทาง: ${explicit}`);
    return path.resolve(explicit);
  }
  const local = path.join(input.home, 'ai-skills');
  return fs.existsSync(local) ? local : null;
}

export function formatResult(r: SyncResult): string {
  const line = (label: string, names: string[]) =>
    `${label} (${names.length}): ${names.length > 0 ? names.join(', ') : '-'}`;
  return [
    line('เพิ่ม', r.added),
    line('อัปเดต', r.updated),
    line('ลบ', r.removed),
    line('ข้าม (ชื่อซ้ำกับ skill ที่เขียนเอง)', r.skipped),
    'ตรวจ diff ใน skills/ แล้ว commit เอง',
  ].join('\n');
}

const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const PLUGIN_DIR = path.join(ROOT, 'skills');

function argFrom(argv: string[]): string | undefined {
  const i = argv.indexOf('--from');
  return i >= 0 ? argv[i + 1] : undefined;
}

function main(): void {
  const config = JSON.parse(fs.readFileSync(path.join(PLUGIN_DIR, 'sync.json'), 'utf8')) as SyncConfig;
  let source = resolveSource({ from: argFrom(process.argv), env: process.env, home: os.homedir() });
  let tmp: string | undefined;
  if (!source) {
    tmp = fs.mkdtempSync(path.join(os.tmpdir(), 'skill-ai-naja-'));
    console.log(`ไม่พบต้นทางในเครื่อง กำลัง clone ${config.repo}`);
    execFileSync('git', ['clone', '--depth', '1', config.repo, tmp], { stdio: 'inherit' });
    source = tmp;
  }
  try {
    console.log(`ต้นทาง: ${source}`);
    const result = syncSkills({
      sourceDir: source,
      targetDir: path.join(PLUGIN_DIR, 'skills'),
      manifestFile: path.join(PLUGIN_DIR, 'synced.json'),
      licensesDir: path.join(PLUGIN_DIR, 'licenses'),
      deny: config.deny,
    });
    console.log(formatResult(result));
  } finally {
    if (tmp) fs.rmSync(tmp, { recursive: true, force: true });
  }
}

try {
  main();
} catch (e) {
  console.error('sync-skills ล้มเหลว:', e instanceof Error ? e.message : e);

  for (const role of ROLE_NAMES) {
    for (const skill of config.roles[role].skills) {
      const name = skill.startsWith('team:') ? skill.slice('team:'.length) : '';
      if (!available.includes(name)) bad.push(`${role}: ${skill}`);
    }
  }
  if (bad.length === 0) return;
  const list = available.length > 0 ? available.map((n) => `team:${n}`).join(', ') : '(ยังไม่มี — รัน npm run sync-skills)';
  throw new Error(`agent-team.config.json เปิด skill ที่ไม่มี: ${bad.join(', ')}\nskill ที่มี: ${list}`);
}

export function loadConfig(
  file: string = path.join(TEAM_ROOT, 'agent-team.config.json'),
  pluginDir: string = SKILLS_PLUGIN_DIR,
): TeamConfig {
  if (!fs.existsSync(file)) return DEFAULT_CONFIG;
  const config = mergeConfig(DEFAULT_CONFIG, JSON.parse(fs.readFileSync(file, 'utf8')));
  checkSkillNames(config, pluginDir);
  return config;
}
