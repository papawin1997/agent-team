// ตรรกะตัวอักษรของ UI ใหม่: grapheme และความกว้างที่ตรงกับ terminal (วิธี A — ผู้ใช้ยืนยันด้วยไม้บรรทัดแล้ว)
// ต่างจาก string-width ของ Ink ตรงสระอำ: "น้ำ" กว้าง 2 ช่อง (น + ำ) ไม่ใช่ 1

const segmenter = new Intl.Segmenter('th', { granularity: 'grapheme' });
const ZERO_WIDTH = /^[\p{Mn}\p{Me}\p{Cf}]$/u;
const PICTOGRAPH = /\p{Extended_Pictographic}/u;

export function graphemes(text: string): string[] {
  return Array.from(segmenter.segment(text), (s) => s.segment);
}

function isWide(cp: number): boolean {
  return (
    (cp >= 0x1100 && cp <= 0x115f) ||
    (cp >= 0x2e80 && cp <= 0xa4cf) ||
    (cp >= 0xac00 && cp <= 0xd7a3) ||
    (cp >= 0xf900 && cp <= 0xfaff) ||
    (cp >= 0xfe30 && cp <= 0xfe4f) ||
    (cp >= 0xff00 && cp <= 0xff60) ||
    (cp >= 0xffe0 && cp <= 0xffe6) ||
    (cp >= 0x20000 && cp <= 0x3fffd)
  );
}

function codePointWidth(ch: string): number {
  const cp = ch.codePointAt(0) ?? 0;
  if (cp < 0x20 || (cp >= 0x7f && cp < 0xa0)) return 0;
  if (ZERO_WIDTH.test(ch)) return 0;
  return isWide(cp) ? 2 : 1;
}

/** จำนวนช่องบนจอ: สระบน/ล่าง วรรณยุกต์ = 0, สระอำ = 1, อักษรจีน/emoji = 2 */
export function displayWidth(text: string): number {
  let width = 0;
  for (const g of graphemes(text)) {
    width += PICTOGRAPH.test(g) ? 2 : Array.from(g).reduce((sum, ch) => sum + codePointWidth(ch), 0);
  }
  return width;
}

/** ขอบ grapheme ก่อนหน้า (cursor ต้องอยู่ที่ขอบ grapheme อยู่แล้ว) */
export function prevBoundary(text: string, cursor: number): number {
  let pos = 0;
  for (const g of graphemes(text.slice(0, cursor))) {
    if (pos + g.length >= cursor) return pos;
    pos += g.length;
  }
  return 0;
}

export function nextBoundary(text: string, cursor: number): number {
  const g = graphemes(text.slice(cursor))[0];
  return g === undefined ? cursor : cursor + g.length;
}

/** Backspace แบบคีย์บอร์ดไทย: ลบทีละ code point */
export function deleteBackward(text: string, cursor: number): { text: string; cursor: number } {
  if (cursor === 0) return { text, cursor };
  const head = Array.from(text.slice(0, cursor));
  head.pop();
  const kept = head.join('');
  return { text: kept + text.slice(cursor), cursor: kept.length };
}

/** offset ในบรรทัดที่ขอบ grapheme ซึ่งความกว้างก่อนหน้าไม่เกิน column (ใช้ตอน ↑↓ ข้ามบรรทัด) */
export function offsetAtWidth(line: string, column: number): number {
  let width = 0;
  let pos = 0;
  for (const g of graphemes(line)) {
    const w = displayWidth(g);
    if (width + w > column) return pos;
    width += w;
    pos += g.length;
  }
  return pos;
}

import { deleteBackward, displayWidth, nextBoundary, offsetAtWidth, prevBoundary } from './text';

/** ปุ่มที่กด: รูปเดียวกับ Key ของ Ink บวกตัวอักษรที่พิมพ์ (paste = ทั้งก้อน) */
export interface KeyInput {
  input: string;
  return?: boolean;
  escape?: boolean;
  tab?: boolean;
  backspace?: boolean;
  delete?: boolean;
  leftArrow?: boolean;
  rightArrow?: boolean;
  upArrow?: boolean;
  downArrow?: boolean;
  ctrl?: boolean;
  meta?: boolean;
}

export interface EditorState {
  value: string;
  /** offset ใน value (อยู่ที่ขอบ grapheme เสมอ) */
  cursor: number;
  /** ข้อความที่เคยส่ง ใหม่สุดก่อน (จำเฉพาะในการรันนี้) */
  history: readonly string[];
  /** -1 = ไม่ได้อยู่ในประวัติ */
  historyIndex: number;
}

export interface EditorResult {
  state: EditorState;
  /** มีค่าเมื่อกด Enter ส่งข้อความ */
  submit?: string;
}

export const emptyEditor = (history: readonly string[] = []): EditorState => ({ value: '', cursor: 0, history, historyIndex: -1 });

const lineStart = (value: string, cursor: number): number => value.lastIndexOf('\n', cursor - 1) + 1;
const lineEnd = (value: string, cursor: number): number => {
  const nl = value.indexOf('\n', cursor);
  return nl === -1 ? value.length : nl;
};

function insert(s: EditorState, input: string): EditorState {
  const text = input.replace(/\r\n?/g, '\n');
  return { ...s, value: s.value.slice(0, s.cursor) + text + s.value.slice(s.cursor), cursor: s.cursor + text.length };
}

/** ข้ามบรรทัดในข้อความหลายบรรทัด (undefined = อยู่บรรทัดแรก/สุดท้ายแล้ว ให้ไปใช้ประวัติแทน) */
function moveLine(s: EditorState, dir: -1 | 1): EditorState | undefined {
  const start = lineStart(s.value, s.cursor);
  const column = displayWidth(s.value.slice(start, s.cursor));
  if (dir === -1) {
    if (start === 0) return undefined;
    const prevStart = lineStart(s.value, start - 1);
    return { ...s, cursor: prevStart + offsetAtWidth(s.value.slice(prevStart, start - 1), column) };
  }
  const end = lineEnd(s.value, s.cursor);
  if (end === s.value.length) return undefined;
  const nextStart = end + 1;
  return { ...s, cursor: nextStart + offsetAtWidth(s.value.slice(nextStart, lineEnd(s.value, nextStart)), column) };
}

function recall(s: EditorState, dir: -1 | 1): EditorState {
  const index = dir === -1 ? Math.min(s.historyIndex + 1, s.history.length - 1) : Math.max(s.historyIndex - 1, -1);
  if (index === s.historyIndex) return s;
  const value = index === -1 ? '' : (s.history[index] ?? '');
  return { ...s, value, cursor: value.length, historyIndex: index };
}

export function editorKey(s: EditorState, k: KeyInput): EditorResult {
  if (k.return) {
    if (s.value.slice(0, s.cursor).endsWith('\\')) {
      return { state: { ...s, value: `${s.value.slice(0, s.cursor - 1)}\n${s.value.slice(s.cursor)}` } };
    }
    const submit = s.value;
    const keep = submit.trim() === '' || s.history[0] === submit;
    return { state: emptyEditor(keep ? s.history : [submit, ...s.history]), submit };
  }
  if (k.escape) return { state: { ...s, value: '', cursor: 0, historyIndex: -1 } };
  if (k.leftArrow) return { state: { ...s, cursor: prevBoundary(s.value, s.cursor) } };
  if (k.rightArrow) return { state: { ...s, cursor: nextBoundary(s.value, s.cursor) } };
  if (k.upArrow) return { state: moveLine(s, -1) ?? recall(s, -1) };
  if (k.downArrow) return { state: moveLine(s, 1) ?? recall(s, 1) };
  if (k.ctrl && k.input === 'a') return { state: { ...s, cursor: lineStart(s.value, s.cursor) } };
  if (k.ctrl && k.input === 'e') return { state: { ...s, cursor: lineEnd(s.value, s.cursor) } };
  if (k.backspace || k.delete) {
    const r = deleteBackward(s.value, s.cursor);
    return { state: { ...s, value: r.text, cursor: r.cursor } };
  }
  if (k.input && !k.ctrl && !k.meta && !k.tab) return { state: insert(s, k.input) };
  return { state: s };
}

import { editorKey, type EditorState, emptyEditor, type KeyInput } from './editor';

export interface MenuOption {
  value: string;
  /** ป้ายภาษาไทย เช่น confirm = ยืนยัน */
  label?: string;
}

export interface MenuState {
  /** แถวที่เลือกอยู่ ; แถวพิมพ์ = options.length */
  index: number;
  /** ข้อความในแถวพิมพ์ (ค้างไว้แม้ย้ายกลับไปที่ตัวเลือก) */
  editor: EditorState;
}

export interface MenuResult {
  state: MenuState;
  choice?: string;
  text?: string;
}

export const newMenu = (): MenuState => ({ index: 0, editor: emptyEditor() });

const printable = (k: KeyInput): boolean => k.input !== '' && !k.ctrl && !k.meta && !k.escape && !k.tab && !k.return;

/** เมนูลูกศร: ↑↓ เลือก, Enter/เลข ยืนยัน, Tab หรือพิมพ์ตัวอักษร = ไปแถวพิมพ์ (allowText) */
export function menuKey(s: MenuState, k: KeyInput, options: readonly MenuOption[], allowText: boolean): MenuResult {
  const rows = options.length + (allowText ? 1 : 0);
  const textRow = allowText ? options.length : -1;
  if (k.upArrow) return { state: { ...s, index: (s.index + rows - 1) % rows } };
  if (k.downArrow) return { state: { ...s, index: (s.index + 1) % rows } };
  if (k.tab && allowText) return { state: { ...s, index: s.index === textRow ? 0 : textRow } };

  if (s.index === textRow) {
    if (k.escape) return { state: s.editor.value !== '' ? { ...s, editor: emptyEditor() } : { ...s, index: 0 } };
    if (k.return && !s.editor.value.slice(0, s.editor.cursor).endsWith('\\')) {
      if (s.editor.value.trim() === '') return { state: s };
      return { state: { ...s, editor: emptyEditor() }, text: s.editor.value };
    }
    return { state: { ...s, editor: editorKey(s.editor, k).state } };
  }

  if (/^[1-9]$/.test(k.input) && !k.ctrl && !k.meta) {
    const option = options[Number(k.input) - 1];
    if (option) return { state: s, choice: option.value };
  }
  if (k.return) {
    const option = options[s.index];
    return option ? { state: s, choice: option.value } : { state: s };
  }
  if (allowText && printable(k)) return { state: { index: textRow, editor: editorKey(s.editor, k).state } };
  return { state: s };
}

/** ข้อความที่พิมพ์ลงจอแล้ว (ลงสีแล้ว) — Ink พิมพ์แบบถาวรด้วย Static */
export interface LogLine {
  id: number;
  text: string;
}

/** agent ที่กำลังทำงาน (จาก StatusSink) */
export interface Activity {
  label: string;
  detail?: string;
  startedAt: number;
}

export type PromptSpec =
  | { kind: 'text'; prompt: string }
  | { kind: 'menu'; prompt: string; options: MenuOption[]; allowText: boolean };
export type PromptRequest = PromptSpec & { id: number };
export type MenuPrompt = Extract<PromptRequest, { kind: 'menu' }>;
export type PromptAnswer = { text: string } | { choice: string };

export interface ScreenState {
  log: readonly LogLine[];
  /** คำถามที่รอคำตอบ (มีได้ทีละคำถาม) */
  prompt?: PromptRequest;
  activity?: Activity;
  /** ข้อความเตือนสั้น ๆ ใต้ช่องพิมพ์ (หายเมื่อมีคำถามใหม่) */
  notice?: string;
  /** ปิดจอแล้ว: เหลือแต่ log */
  closed?: boolean;
}

export interface Mounted {
  unmount(): void;
}

/** สถานะจอที่ TuiIO เขียนและ App (Ink) อ่าน — แยกจาก Ink เพื่อให้เทสต์ได้โดยไม่ต้องมี terminal */
export class ScreenStore {
  private state: ScreenState = { log: [] };
  private readonly listeners = new Set<() => void>();
  private nextId = 1;
  private pending: { resolve: (answer: PromptAnswer) => void; reject: (error: Error) => void } | undefined;

  readonly getState = (): ScreenState => this.state;

  readonly subscribe = (listener: () => void): (() => void) => {
    this.listeners.add(listener);
    return () => {
      this.listeners.delete(listener);
    };
  };

  print(text: string): void {
    this.set({ log: [...this.state.log, { id: this.nextId++, text }] });
  }

  setActivity(activity: Activity | undefined): void {
    this.set({ activity });
  }

  setNotice(notice: string | undefined): void {
    this.set({ notice });
  }

  request(spec: PromptSpec): Promise<PromptAnswer> {
    if (this.state.closed) return Promise.reject(new Error('หน้าจอปิดแล้ว'));
    if (this.pending) return Promise.reject(new Error('มีคำถามค้างอยู่แล้ว'));
    return new Promise<PromptAnswer>((resolve, reject) => {
      this.pending = { resolve, reject };
      this.set({ prompt: { ...spec, id: this.nextId++ }, notice: undefined });
    });
  }

  answer(answer: PromptAnswer): void {
    const pending = this.pending;
    if (!pending) return;
    this.pending = undefined;
    this.set({ prompt: undefined });
    pending.resolve(answer);
  }

  close(error: Error): void {
    const pending = this.pending;
    this.pending = undefined;
    this.set({ prompt: undefined, activity: undefined, notice: undefined, closed: true });
    pending?.reject(error);
  }

  private set(patch: Partial<ScreenState>): void {
    this.state = { ...this.state, ...patch };
    for (const listener of this.listeners) listener();
  }
}

  /** เปิดจอ (index ใช้ mountApp ของ Ink ; เทสต์ส่งตัวปลอม) */
  mount: (store: ScreenStore) => Mounted;
  now?: () => number;
}

/** prompt ของ readline มี "> " ท้าย — UI ใหม่มีช่องพิมพ์อยู่แล้วจึงตัดออก */
export function cleanPrompt(prompt: string): string {
  return prompt.replace(/\s*>\s*$/, '').trimEnd();
}

export function menuOptions(options: readonly string[]): MenuOption[] {
  return options.map((value) => {
    const label = CHOICE_LABELS[value];
    return label ? { value, label } : { value };
  });
}

/** UserIO + StatusSink ของ UI ใหม่: เขียนสถานะลง ScreenStore และรอคำตอบจากจอ */
export class TuiIO implements UserIO {
  readonly store: ScreenStore;
  readonly status: StatusSink;
  private readonly s: Styler;
  private readonly mounted: Mounted;
  private closed = false;

  constructor(o: TuiIOOptions) {
    this.store = o.store ?? new ScreenStore();
    this.s = styler(o.color ?? true);
    const now = o.now ?? Date.now;
    this.status = {
      start: (label) => this.store.setActivity({ label, startedAt: now() }),
      update: (detail) => {
        const current = this.store.getState().activity;
        if (current) this.store.setActivity({ ...current, detail });
      },
      stop: () => this.store.setActivity(undefined),
    };
    this.mounted = o.mount(this.store);
  }

  say(text: string, kind?: SayKind): void {
    this.store.print(styleSay(text, kind, this.s));
  }

  async ask(prompt: string): Promise<string> {
    const answer = await this.request({ kind: 'text', prompt: cleanPrompt(prompt) });
    const text = ('text' in answer ? answer.text : answer.choice).trim();
    this.store.print(this.s.dim(`> ${text}`));
    return text;
  }

  async chooseOrText<T extends string>(prompt: string, options: readonly T[]): Promise<T | { text: string }> {
    const answer = await this.request({ kind: 'menu', prompt: cleanPrompt(prompt), options: menuOptions(options), allowText: true });
    if ('choice' in answer) return this.chosen(answer.choice as T);
    const text = answer.text.trim();
    this.store.print(this.s.dim(`> ${text}`));
    return { text };
  }

  async choose<T extends string>(prompt: string, options: readonly T[]): Promise<T> {
    const answer = await this.request({ kind: 'menu', prompt: cleanPrompt(prompt), options: menuOptions(options), allowText: false });
    if (!('choice' in answer)) throw new Error('คำถามนี้ต้องเลือกจากตัวเลือกเท่านั้น');
    return this.chosen(answer.choice as T);
  }

  close(): void {
    if (this.closed) return;
    this.closed = true;
    this.store.close(new Error(EOF_MESSAGE));
    this.mounted.unmount();
  }

  private chosen<T extends string>(choice: T): T {
    const label = CHOICE_LABELS[choice];
    this.store.print(this.s.dim(`เลือก: ${label ?? choice} (${choice})`));
    return choice;
  }

  private request(spec: PromptSpec): Promise<PromptAnswer> {
    if (this.closed) return Promise.reject(new Error(EOF_MESSAGE));
    return this.store.request(spec);
  }
}

function withCursor(line: string, at: number): string {
  const g = graphemes(line.slice(at))[0];
  const after = g ? line.slice(at + g.length) : '';
  return `${line.slice(0, at)}${INVERSE}${g ?? ' '}${INVERSE_OFF}${after}`;
}

function EditorLines({ editor, active, prefix }: { editor: EditorState; active: boolean; prefix: string }): React.ReactElement {
  const lines = editor.value.split('\n');
  const indent = ' '.repeat(Array.from(prefix).length);
  let start = 0;
  return h(
    Box,
    { flexDirection: 'column' },
    ...lines.map((line, i) => {
      const lineStart = start;
      start += line.length + 1;
      const here = active && editor.cursor >= lineStart && editor.cursor <= lineStart + line.length;
      const body = here ? withCursor(line, editor.cursor - lineStart) : line;
      return h(Text, { key: i }, `${i === 0 ? prefix : indent}${body}`);
    }),
  );
}

  // เก็บสถานะช่องพิมพ์/เมนูใน ref: ปุ่มที่มาติดกันก่อน React วาดใหม่ (พิมพ์เร็ว) ต้องเห็นค่าล่าสุดเสมอ

      prompt?.kind === 'menu' ? null : h(Text, { dimColor: true }, HINT),
    ),
  );
}

  return o.stdinTTY && o.stdoutTTY && o.rawMode ? 'tui' : 'classic';
}

export function createTerminalIO(o: { env?: NodeJS.ProcessEnv; warn?: (message: string) => void } = {}): TerminalIO {
  const env = o.env ?? process.env;
  const kind = pickUi({
    stdinTTY: process.stdin.isTTY === true,
    stdoutTTY: process.stdout.isTTY === true,
    rawMode: typeof process.stdin.setRawMode === 'function',
    env,
  });
  if (kind === 'tui') {
    try {
      return new TuiIO({
        color: colorEnabled(process.stdout, env),
        // Ctrl+C ส่งต่อเป็น SIGINT เหมือน CliIO ให้ handler หยุดงานใน index ทำงาน
        mount: (store) => mountApp(store, () => process.emit('SIGINT')),
      });
    } catch (e) {
      const warn = o.warn ?? ((message: string) => console.error(message));
      warn(`เปิดหน้าจอแบบใหม่ไม่สำเร็จ (${e instanceof Error ? e.message : String(e)}) — ใช้หน้าจอแบบเดิมแทน`);
    }
  }
  return new CliIO();
}
